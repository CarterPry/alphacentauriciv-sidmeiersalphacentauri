13909: bool JitWasmCodeGen::shouldStopBlockBefore(U32 eip, DecodedOp* op) {
13910: #if !defined(BOXEDWINE_MULTI_THREADED) && !defined(__TEST)
13911:     // Honor profile-guided split hints from the grouped manifest: end the
13912:     // block before a hot interior target so that target compiles (and is
13913:     // cached) as its own block-start entry, making it reachable by the
13914:     // grouped direct-call machinery. The JS side keeps the hint map; without
13915:     // an active persistence session there are no hints and no EM_JS call.
13916:     if (!wasmJitPersistenceActive() || !op || eip == this->startingEip) {
13917:         return false;
13918:     }
13919:     if (boxedwine_wasm_profile_split_before(
13920:         this->startingEip - cpu->seg[CS].address,
13921:         eip - cpu->seg[CS].address) != 0) {
13922:         m_profileSplitTargetOp = op;
13923:         m_profileSplitBlockStartEip = this->startingEip;
13924:         m_profileSplitTargetEip = eip;
13925:         return true;
13926:     }
13927:     return false;
13928: #else
